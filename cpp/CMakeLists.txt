cmake_minimum_required(VERSION 3.16)
project(btc_event_hft_v2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(hft_core src/timestamp.cpp src/orderbook.cpp src/event_engine.cpp src/risk_engine.cpp src/paper_execution.cpp)
target_include_directories(hft_core PUBLIC include)
enable_testing()
add_executable(hft_tests tests/test_core.cpp)
target_link_libraries(hft_tests PRIVATE hft_core)
add_test(NAME hft_tests COMMAND hft_tests)
